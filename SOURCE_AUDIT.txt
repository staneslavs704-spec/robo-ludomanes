@router.message(CommandStart())
async def start(message: Message, bot: Bot):
    args = message.text.split(maxsplit=1)
    start_param = args[1].strip() if len(args) > 1 else ""
    async with SessionLocal() as session:
        user = await ensure_user(session, {
            "id": message.from_user.id,
            "username": message.from_user.username,
            "first_name": message.from_user.first_name,
            "language_code": message.from_user.language_code,
        }, start_param)
    keyboard = None
    if settings.webapp_url.startswith("https://") and "example.com" not in settings.webapp_url:
        keyboard = InlineKeyboardMarkup(inline_keyboard=[[
            InlineKeyboardButton(text="🚀 Открыть ROBO LUDOMANES", web_app=WebAppInfo(url=settings.webapp_url))
        ]])
    await message.answer(
        f"Привет, <b>{escape(message.from_user.first_name or 'игрок')}!</b>\n\n"
        "Добро пожаловать в ROBO LUDOMANES. Открой мини-приложение, чтобы посмотреть баланс, задания и рулетки.\n\n"
        f"Твои рефералы: <b>{user.referrals_count}</b>\n"
        f"Бесплатные прокрутки: <b>{user.free_spins}</b>",
        reply_markup=keyboard, parse_mode="HTML"
    )

@router.pre_checkout_query()
async def pre_checkout(query: PreCheckoutQuery, bot: Bot):
    async with SessionLocal() as session:
        payment = await session.scalar(select(Payment).where(Payment.payload == query.invoice_payload))
        valid = bool(settings.paid_wheel_enabled and payment and payment.status == "pending" and payment.user_id == query.from_user.id
                     and payment.amount == query.total_amount and query.currency == "XTR")
    await bot.answer_pre_checkout_query(
        query.id, ok=valid,
        error_message=None if valid else "Счёт недействителен или уже обработан. Создайте новый счёт."
    )


@router.message(F.successful_payment)
async def successful_payment(message: Message):
    sp = message.successful_payment
    async with SessionLocal() as session:
        payment = await session.scalar(select(Payment).where(Payment.payload == sp.invoice_payload).with_for_update())
        if not payment:
            log.error("Payment payload not found: %s", sp.invoice_payload)
            await message.answer("Платёж получен, но заказ не найден. Напишите /paysupport.")
            return
        if payment.status != "pending":
            await message.answer("Этот платёж уже обработан или возвращён. Повторного зачисления не будет.")
            return
        if payment.user_id != message.from_user.id or payment.amount != sp.total_amount or sp.currency != "XTR":
            log.error("Payment data mismatch for payment %s", payment.id)
            await message.answer("Не удалось автоматически сопоставить платёж. Напишите /paysupport.")
            return
        user = await session.scalar(select(User).where(User.id == payment.user_id).with_for_update())
        if not user:
            await message.answer("Профиль не найден. Напишите /paysupport.")
            return
        payment.status = "paid"
        payment.charge_id = sp.telegram_payment_charge_id
        payment.paid_at = datetime.now(timezone.utc)
        user.paid_spins += 1
        try:
            await session.commit()
        except IntegrityError:
            await session.rollback()
            log.exception("Duplicate Stars charge detected")
            await message.answer("Платёж уже обрабатывается. При необходимости напишите /paysupport.")
            return
    await message.answer("✅ Оплата подтверждена! Одна платная прокрутка добавлена. Открой мини-приложение и обнови баланс.")

@router.message(Command("refund"))
async def refund_payment(message: Message, bot: Bot):
    if message.from_user.id != settings.admin_id:
        return
    raw = message.text.partition(" ")[2].strip()
    if not raw.isdigit():
        await message.answer("Формат: /refund ID_ПЛАТЕЖА (ID из /payments)")
        return
    async with SessionLocal() as session:
        hint = await session.get(Payment, int(raw))
        if not hint:
            await message.answer("Платёж не найден.")
            return
        # Lock order matches paid-spin flow: user first, then payment.
        user = await session.scalar(select(User).where(User.id == hint.user_id).with_for_update())
        payment = await session.scalar(select(Payment).where(Payment.id == int(raw)).with_for_update())
        if not payment or payment.status != "paid" or not payment.charge_id:
            await message.answer("Оплаченный платёж не найден или уже обработан.")
            return
        if payment.consumed_at is not None:
            await message.answer("Эта прокрутка уже использована. Автоматический возврат заблокирован, чтобы не возвращать оплату за уже выданную услугу.")
            return
        try:
            await bot.refund_star_payment(user_id=payment.user_id, telegram_payment_charge_id=payment.charge_id)
        except Exception:
            log.exception("Telegram Stars refund failed")
            await message.answer("Telegram не подтвердил возврат. Проверьте лог и статус платежа.")
            return
        payment.status = "refunded"
        if user and user.paid_spins > 0:
            user.paid_spins -= 1
        await session.commit()
    await message.answer(f"✅ Возврат по платежу #{raw} отправлен в Telegram.")

        await message.answer("Формат: /fulfill ID_ЗАЯВКИ")

        log.exception("Could not notify user about fulfilled NFT claim")
    await message.answer(f"✅ Заявка #{raw} отмечена как выполненная.")


async def main():
    if not settings.is_configured:
        raise RuntimeError("Заполните BOT_TOKEN и ADMIN_ID в .env")
    await init_db()
    bot = Bot(settings.bot_token)
    dp = Dispatcher()
    dp.include_router(router)
    log.info("ROBO LUDOMANES bot started")
    try:
        await dp.start_polling(bot, allowed_updates=dp.resolve_used_update_types())
    finally:
        await bot.session.close()

class Settings(BaseSettings):
    bot_token: str = ""
    admin_id: int = 0
    webapp_url: str = "https://example.com"
    database_url: str = "sqlite+aiosqlite:///./database.db"

    quest_spin_cost: int = 100

    points_per_referral: int = 10
    points_for_channel: int = 50
    required_channel: str = ""
    channel_url: str = ""
    init_data_max_age_seconds: int = 86400
    log_level: str = "INFO"

    model_config = SettingsConfigDict(env_file=".env", env_file_encoding="utf-8", extra="ignore")

    @property
    def is_configured(self) -> bool:
        return bool(self.bot_token and ":" in self.bot_token and self.admin_id > 0)


@lru_cache

    return Settings()

    "pool_pre_ping": True,
    "connect_args": connect_args,
}

if settings.database_url.startswith("sqlite"):

        cursor = dbapi_connection.cursor()
        cursor.execute("PRAGMA foreign_keys=ON")
        cursor.execute("PRAGMA journal_mode=WAL")
        cursor.execute("PRAGMA busy_timeout=5000")
        cursor.close()

import secrets
from datetime import datetime, timezone
from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.exc import IntegrityError
from app.models import User, Spin, NftInventory, NftClaim, Payment
from app.config import get_settings

settings = get_settings()


async def ensure_user(session: AsyncSession, tg_user: dict, start_param: str = "") -> User:
    user_id = int(tg_user["id"])
    user = await session.get(User, user_id)
    if user:
        user.username = tg_user.get("username")
        user.first_name = tg_user.get("first_name") or user.first_name
        user.language_code = tg_user.get("language_code")
        await session.commit()
        return user

    referrer_id = None
    if start_param.startswith("ref_"):
        try:
            candidate = int(start_param[4:])
            if candidate != user_id and await session.get(User, candidate):
                referrer_id = candidate
        except (TypeError, ValueError):
            pass
    user = User(
        id=user_id,
        username=tg_user.get("username"),
        first_name=tg_user.get("first_name") or "Игрок",
        language_code=tg_user.get("language_code"),
        referred_by=referrer_id,
    )
    session.add(user)
    try:
        if referrer_id:
            referrer = await session.get(User, referrer_id, with_for_update=True)
            if referrer:
                referrer.referrals_count += 1
                referrer.quest_points += settings.points_per_referral
                eligible = referrer.referrals_count // max(1, settings.referrals_per_free_spin)
                bonus = eligible - referrer.free_spins_granted
                if bonus > 0:
                    referrer.free_spins += bonus
                    referrer.free_spins_granted += bonus
        await session.commit()
    except IntegrityError:
        await session.rollback()
        user = await session.get(User, user_id)
        if user:
            return user
        raise
    return user


async def execute_spin(session: AsyncSession, user_id: int, wheel_type: str, request_id: str) -> dict:
    # Idempotency: retrying the same request returns the original result without another debit.
    existing = await session.scalar(select(Spin).where(Spin.user_id == user_id, Spin.request_id == request_id))
    if existing:
        claim_id = None
        if existing.nft_id:
            claim = await session.scalar(select(NftClaim).where(NftClaim.spin_id == existing.id))
            claim_id = claim.id if claim else None
        return _spin_result(existing, claim_id, replay=True)

    user = await session.scalar(select(User).where(User.id == user_id).with_for_update())
    if not user or user.is_blocked:
        raise ValueError("Профиль недоступен")
    # A second check after taking the user-row lock serializes duplicate concurrent requests.
    existing = await session.scalar(select(Spin).where(Spin.user_id == user_id, Spin.request_id == request_id))
    if existing:
        claim_id = None
        if existing.nft_id:
            claim = await session.scalar(select(NftClaim).where(NftClaim.spin_id == existing.id))
            claim_id = claim.id if claim else None
        return _spin_result(existing, claim_id, replay=True)
    payment_credit = None
    if wheel_type == "free":
        if user.free_spins < 1:
            raise ValueError("Нет бесплатных прокруток. Пригласите друзей.")
        user.free_spins -= 1
    elif wheel_type == "paid":
        if not settings.paid_wheel_enabled:
            raise ValueError("Платная рулетка временно отключена")
        if user.paid_spins < 1:
            raise ValueError("Сначала приобретите платную прокрутку за Telegram Stars.")
        payment_credit = await session.scalar(
            select(Payment).where(
                Payment.user_id == user_id, Payment.status == "paid", Payment.consumed_at.is_(None)
            ).order_by(Payment.created_at).limit(1).with_for_update()
        )
        if not payment_credit:
            raise ValueError("Не удалось сопоставить купленную попытку с платежом. Обратитесь в поддержку.")
        payment_credit.consumed_at = datetime.now(timezone.utc)
        user.paid_spins -= 1
    elif wheel_type == "quest":
        if user.quest_points < settings.quest_spin_cost:
            raise ValueError(f"Для этой рулетки нужно {settings.quest_spin_cost} баллов.")
        user.quest_points -= settings.quest_spin_cost
    else:
        raise ValueError("Неизвестная рулетка")

    # Original game balance: 56% empty, 40% points, 4% NFT (only if stock exists).
    nft_count = await session.scalar(select(func.count(NftInventory.id)).where(NftInventory.status == "available"))
    outcomes = [("empty", "В этот раз без приза", 56),
                ("points", "10 баллов", 20),
                ("points", "25 баллов", 15),
                ("points", "50 баллов", 5)]
    if nft_count:
        outcomes.append(("nft", "NFT-подарок", 4))
    prize_type, title, weight = secrets.SystemRandom().choices(outcomes, weights=[x[2] for x in outcomes], k=1)[0]
    points = 0
    nft_id = None
    if prize_type == "points":
        points = int(title.split()[0])
        user.quest_points += points
    elif prize_type == "nft":
        # Lock only the selected inventory row, not the whole stock, to reduce contention.
        offset = secrets.randbelow(max(1, nft_count))
        nft = await session.scalar(
            select(NftInventory).where(NftInventory.status == "available")
            .order_by(NftInventory.id).offset(offset).limit(1).with_for_update(skip_locked=True)
        )
        if not nft:
            nft = await session.scalar(
                select(NftInventory).where(NftInventory.status == "available")
                .order_by(NftInventory.id).limit(1).with_for_update(skip_locked=True)
            )
        if nft:
            nft.status = "reserved"
            nft_id = nft.id
            title = nft.title
        else:
            # All stock may have been reserved by concurrent winners.
            prize_type, title, points = "empty", "В этот раз без приза", 0

    spin = Spin(user_id=user_id, request_id=request_id, wheel_type=wheel_type,
                prize_type=prize_type, prize_title=title, points_awarded=points, nft_id=nft_id,
                payment_id=payment_credit.id if payment_credit else None)
    session.add(spin)
    await session.flush()
    claim_id = None
    if nft_id:
        claim = NftClaim(user_id=user_id, spin_id=spin.id, nft_id=nft_id)
        session.add(claim)
        await session.flush()
        claim_id = claim.id
    await session.commit()
    return _spin_result(spin, claim_id, replay=False)


def _spin_result(spin: Spin, claim_id: int | None, replay: bool) -> dict:
    return {
        "spin_id": spin.id,
        "wheel_type": spin.wheel_type,
        "prize_type": spin.prize_type,
        "title": spin.prize_title,
        "points_awarded": spin.points_awarded,
        "claim_id": claim_id,
        "replayed": replay,
    }

@asynccontextmanager
async def lifespan(app: FastAPI):
    await init_db()
    app.state.bot = Bot(settings.bot_token) if settings.is_configured else None
    app.state.bot_username = ""
    if app.state.bot:
        try:
            app.state.bot_username = (await app.state.bot.get_me()).username or ""
        except Exception:
            log.exception("Could not resolve bot username during startup")
    yield
    if app.state.bot:
        await app.state.bot.session.close()

async def current_user(

    session: AsyncSession = Depends(get_session),

async def claim_task(payload: TaskClaimRequest, request: Request,
                     user: User = Depends(current_user), session: AsyncSession = Depends(get_session)):
    if payload.task_key != "subscribe_channel" or not settings.required_channel:
        raise HTTPException(status_code=400, detail="Это задание сейчас недоступно")
    existing = await session.scalar(select(TaskClaim).where(
        TaskClaim.user_id == user.id, TaskClaim.task_key == payload.task_key
    ))
    if existing:
        raise HTTPException(status_code=409, detail="Награда за это задание уже получена")
    bot: Bot | None = request.app.state.bot
    if not bot:
        raise HTTPException(status_code=503, detail="Бот не настроен")
    try:
        member = await bot.get_chat_member(settings.required_channel, user.id)
        is_member = member.status in {"member", "administrator", "creator"} or (member.status == "restricted" and getattr(member, "is_member", False))
        if not is_member:
            raise HTTPException(status_code=400, detail="Сначала подпишитесь на канал")
    except HTTPException:
        raise
    except Exception as exc:
        log.exception("Channel membership verification failed")
        raise HTTPException(status_code=502, detail="Не удалось проверить подписку. Проверьте права бота в канале.") from exc
    locked_user = await session.scalar(select(User).where(User.id == user.id).with_for_update())
    # Unique constraint protects against concurrent duplicate claims.
    session.add(TaskClaim(user_id=user.id, task_key=payload.task_key, points_awarded=settings.points_for_channel))
    locked_user.quest_points += settings.points_for_channel
    try:
        await session.commit()
    except Exception as exc:
        await session.rollback()
        raise HTTPException(status_code=409, detail="Задание уже засчитано") from exc
    return {"ok": True, "points_awarded": settings.points_for_channel}

import hashlib
import hmac
import json
import time
from urllib.parse import parse_qsl
from fastapi import HTTPException, Header
from app.config import get_settings


def validate_init_data(init_data: str, bot_token: str, max_age: int) -> dict:
    """Validate Telegram WebApp initData according to Telegram's HMAC algorithm."""
    if not init_data or len(init_data) > 8192:
        raise HTTPException(status_code=401, detail="Нет данных авторизации Telegram")
    try:
        pairs = parse_qsl(init_data, keep_blank_values=True, strict_parsing=True)
        data = dict(pairs)
        if len(data) != len(pairs):
            raise ValueError("duplicate keys")
        received_hash = data.pop("hash", None)
        if not received_hash:
            raise ValueError("hash missing")
        check_string = "\n".join(f"{k}={v}" for k, v in sorted(data.items()))
        secret_key = hmac.new(b"WebAppData", bot_token.encode(), hashlib.sha256).digest()
        calculated_hash = hmac.new(secret_key, check_string.encode(), hashlib.sha256).hexdigest()
        if not hmac.compare_digest(calculated_hash, received_hash):
            raise ValueError("signature mismatch")
        auth_date = int(data.get("auth_date", "0"))
        now = int(time.time())
        if auth_date <= 0 or auth_date > now + 60 or now - auth_date > max_age:
            raise ValueError("initData expired")
        user = json.loads(data.get("user", "{}"))
        if not isinstance(user, dict) or not isinstance(user.get("id"), int):
            raise ValueError("user missing")
        return {"user": user, "auth_date": auth_date, "start_param": data.get("start_param", "")}
    except HTTPException:
        raise
    except Exception as exc:
        raise HTTPException(status_code=401, detail="Недействительная или устаревшая авторизация Telegram") from exc


async def telegram_user(x_telegram_init_data: str | None = Header(default=None)) -> dict:
    settings = get_settings()
    if not settings.is_configured:
        raise HTTPException(status_code=503, detail="Сначала настройте BOT_TOKEN и ADMIN_ID в .env")
    return validate_init_data(x_telegram_init_data or "", settings.bot_token, settings.init_data_max_age_seconds)

import hashlib
import hmac
import json
import time
import unittest
from urllib.parse import urlencode
from app.security import validate_init_data


class TelegramInitDataTests(unittest.TestCase):
    def make_init_data(self, token: str, user_id: int):
        data = {
            "auth_date": str(int(time.time())),
            "user": json.dumps({"id": user_id, "first_name": "Test"}, separators=(",", ":")),
        }
        check = "\n".join(f"{k}={v}" for k, v in sorted(data.items()))
        secret = hmac.new(b"WebAppData", token.encode(), hashlib.sha256).digest()
        data["hash"] = hmac.new(secret, check.encode(), hashlib.sha256).hexdigest()
        return urlencode(data)

    def test_valid_data(self):
        token = "123456:TEST_TOKEN"
        result = validate_init_data(self.make_init_data(token, 12345), token, 86400)
        self.assertEqual(result["user"]["id"], 12345)

    def test_tampered_data_rejected(self):
        token = "123456:TEST_TOKEN"
        raw = self.make_init_data(token, 12345).replace("12345", "99999")
        with self.assertRaises(Exception):
            validate_init_data(raw, token, 86400)

# ROBO LUDOMANES — новая серверная основа

Проект разделён на Telegram-бота, Mini App и API. Локально можно проверять на SQLite в Termux. Для публичного запуска используйте VPS и PostgreSQL через Docker Compose.

## Что реализовано

- Авторизация Mini App через криптографическую проверку `Telegram.WebApp.initData`; переданный браузером `user_id` не используется как источник доверия.
- Регистрация пользователей и рефералов через `/start ref_TELEGRAM_ID`.
- Баллы за действительно нового пользователя; одна регистрация засчитывается только один раз.
- Бесплатная прокрутка за каждые 3 приглашения (число настраивается).
- Три рулетки: бесплатная, за баллы и платная.
- Результат, баланс, списания и призы определяются сервером.
- Идемпотентность прокруток по `request_id`, чтобы повторный сетевой запрос не списал попытку второй раз.
- Платная попытка покупается через Telegram Stars (`XTR`). Mini App открывает нативный счёт Telegram, а попытка начисляется только после `successful_payment`, а не после нажатия кнопки оплаты. Администратор может посмотреть платежи через `/payments` и вернуть неиспользованную попытку командой `/refund ID`.
- Задание подписки проверяется через `getChatMember`; бот должен быть администратором канала.
- NFT-приз выпадает только из фактически добавленного запаса подарков. Победа создаёт персональную заявку; произвольный ID чужого подарка принять нельзя.
- Админ-команды `/stats`, `/addgift`, `/giftstock`, `/fulfill`, `/payments`, `/refund`.
- `/paysupport` для обращений по оплатам.

## 1. Подготовка в Termux

Установите Python и системные инструменты. Для автоматической установки зависимостей после распаковки можно запустить `bash install_termux.sh`; ниже приведён ручной вариант:

```bash
pkg update -y && pkg upgrade -y
pkg install python git nano clang libffi openssl -y
```

Распакуйте проект, перейдите в папку `robo_ludomanes` и выполните:

```bash
python -m venv .venv
source .venv/bin/activate
pip install --upgrade pip setuptools wheel
pip install -r requirements.txt
cp .env.example .env
nano .env
```

Откройте бота в Telegram и отправьте ему `/start` (это также позволит боту писать администратору об NFT-заявках и оплатах). Заполните как минимум:

- `BOT_TOKEN` — токен из BotFather;
- `ADMIN_ID` — ваш Telegram ID числом;
- `WEBAPP_URL` — публичный HTTPS-адрес Mini App. Для локального теста без публичного HTTPS можно проверять API, но Telegram Mini App не откроется с обычного `http://` адреса.

Локальная установка через `requirements.txt` использует SQLite и не требует компиляции драйвера PostgreSQL. Для публичного VPS-деплоя Docker дополнительно устанавливает `asyncpg` через `requirements-vps.txt`.

Для локального запуска оставьте SQLite:

```env
DATABASE_URL=sqlite+aiosqlite:///./database.db
```

## 2. Запуск в Termux

Нужны два процесса — API и бот. Откройте две сессии Termux.

**Сессия №1 — API:**

```bash
cd ~/robo_ludomanes
source .venv/bin/activate
uvicorn app.main:app --host 0.0.0.0 --port 8000
```

**Сессия №2 — бот:**

```bash
cd ~/robo_ludomanes
source .venv/bin/activate
python bot.py
```

Проверка API: откройте `http://127.0.0.1:8000/health`. Ожидаемый ответ: `{"status":"ok","service":"robo-ludomanes"}`. Для проверки базы данных также доступен `/ready`.

Для проверки Mini App в самом Telegram нужен доступный из интернета HTTPS URL. Не публикуйте порт Termux напрямую без HTTPS и не рассчитывайте на Serveo как на постоянную инфраструктуру. Для временного теста используйте HTTPS-туннель, а для запуска на аудиторию — VPS с доменом и TLS.

Не добавляйте в запас подарки, которыми проект не располагает.

## 5. Переход на VPS

1. Направьте домен на VPS и настройте HTTPS через Caddy, Nginx + Certbot или аналогичный reverse proxy.
2. Установите Docker Engine и Docker Compose plugin.
3. Создайте `.env` на VPS на основе `.env.example`.
4. В `WEBAPP_URL` укажите HTTPS адрес Mini App.
5. Установите `POSTGRES_PASSWORD` в случайный длинный пароль (например, результат `openssl rand -hex 32`).
6. Запустите:

```bash
docker compose up -d --build
```

7. Reverse proxy должен проксировать HTTPS-трафик на `127.0.0.1:8000`.
8. Проверьте `https://ВАШ-ДОМЕН/health`, затем откройте `/start` у бота.

Пример значения `POSTGRES_PASSWORD` генерируйте отдельно, например `openssl rand -hex 32`. Не используйте примерный или простой пароль.

## 6. Правила игры и настройки

| Переменная | Назначение | По умолчанию |
|---|---|---:|
| `STARS_PER_SPIN` | Цена одной платной попытки | 50 Stars |
| `PAID_WHEEL_ENABLED` | Разрешить платную рулетку после правовой проверки | false |
| `QUEST_SPIN_COST` | Стоимость рулетки за баллы | 100 баллов |
| `REFERRALS_PER_FREE_SPIN` | Приглашений на бесплатную попытку | 3 |
| `POINTS_PER_REFERRAL` | Баллы за нового реферала | 10 |
| `POINTS_FOR_CHANNEL` | Баллы за подписку | 50 |

Платная рулетка по умолчанию отключена (`PAID_WHEEL_ENABLED=false`). Включайте её только после проверки местных законов, правил Telegram и юридической модели призов. Если механика с оплатой и случайными призами подпадает под регулирование азартных игр или лотерей, одного технического решения недостаточно. После согласования можно установить `PAID_WHEEL_ENABLED=true`.

Текущая таблица шансов при наличии подарков: 56 весовых долей — пустой результат, 40 — баллы (10/25/50 с разными весами), 4 — NFT. Если запас NFT пуст, NFT-сектор не участвует в розыгрыше.

## 7. Проверка перед публичным запуском

Обязательно проведите тестовые оплаты через тестовое окружение Telegram Stars, тесты повторных запросов, одновременных прокруток, подписки на канал, возвратов и восстановления базы из резервной копии.

## Важно перед запуском на 5–7 тысяч пользователей

Эта сборка — рабочая стартовая версия с базовыми защитами, а не заверенный аудитом финансовый продукт. До публикации необходимо:

- заменить тестовый токен, который ранее находился в исходном коде, через BotFather; хранить новый только в `.env`/секретах VPS;
- поставить HTTPS и домен; не запускать публичный сервис через Flask development server или постоянный Serveo-туннель;
- настроить ежедневные резервные копии PostgreSQL и проверить восстановление;
- добавить миграции схемы БД (Alembic) до дальнейших обновлений;
- провести нагрузочное тестирование с профилем одновременных пользователей, а не только по общему количеству регистраций;
- проверить условия розыгрыша, правила возвратов, поддержку платежей и юридические требования к механике призов в регионе запуска.

В PostgreSQL выставлен пул соединений API/бота (10 основных + до 20 дополнительных соединений на процесс). У проекта нет Redis, потому что для базовой нагрузки 5–7 тысяч зарегистрированных пользователей он не обязателен. Его можно добавить при реальном росте нагрузки — например для распределённого rate limiting и фоновых задач.

## 8. Резервное копирование и кнопка Mini App

На VPS резервную копию PostgreSQL можно создать так:

```bash
bash backup_db.sh
```

Для ежедневного запуска настройте cron на VPS и храните копии ещё и вне самого сервера. Периодически проверяйте восстановление из копии. Не используйте `docker compose down -v`, если не хотите удалить том с базой данных.

После того как домен и HTTPS готовы, в BotFather можно дополнительно настроить Menu Button: `/setmenubutton` → выбрать бота → указать текст кнопки и `WEBAPP_URL`. Основная кнопка открытия приложения уже отправляется командой `/start`.

Файл `Caddyfile.example` — минимальный пример reverse proxy для Caddy на VPS. Замените `your-domain.example` на свой домен; DNS должен вести на VPS, а входящие порты 80 и 443 должны быть доступны.
